<script>
    function enterWorkspace() {
        const overlay = document.getElementById('welcome-overlay');
        const workspace = document.getElementById('main-workspace');
        
        // Start transition
        overlay.style.opacity = '0';
        overlay.style.transform = 'scale(1.05)';
        
        setTimeout(() => {
            overlay.style.visibility = 'hidden';
            workspace.classList.add('visible-workspace');
        }, 600);
    }

    function switchTab(tab) {
        const codeView = document.getElementById('view-code');
        const graphView = document.getElementById('view-graph');
        const tabCodeBtn = document.getElementById('tab-code');
        const tabGraphBtn = document.getElementById('tab-graph');
        const codeIndicator = document.getElementById('active-indicator-code');
        const graphIndicator = document.getElementById('active-indicator-graph');

        if (tab === 'code') {
            // UI Classes
            codeView.classList.remove('opacity-0', 'pointer-events-none');
            codeView.classList.add('opacity-100');
            graphView.classList.add('opacity-0', 'pointer-events-none');
            graphView.classList.remove('opacity-100');
            
            // Buttons
            tabCodeBtn.className = "px-6 flex items-center gap-2 bg-white/80 border-r border-border/50 text-nude-900 text-[11px] font-label uppercase tracking-wider font-semibold relative transition-all";
            tabGraphBtn.className = "px-6 flex items-center gap-2 text-nude-400 hover:bg-white/40 text-[11px] font-label uppercase tracking-wider transition-all border-r border-border/50";
            
            codeIndicator.classList.remove('hidden');
            graphIndicator.classList.add('hidden');
        } else {
            // UI Classes
            graphView.classList.remove('opacity-0', 'pointer-events-none');
            graphView.classList.add('opacity-100');
            codeView.classList.add('opacity-0', 'pointer-events-none');
            codeView.classList.remove('opacity-100');
            
            // Buttons
            tabGraphBtn.className = "px-6 flex items-center gap-2 bg-white/80 border-r border-border/50 text-nude-900 text-[11px] font-label uppercase tracking-wider font-semibold relative transition-all";
            tabCodeBtn.className = "px-6 flex items-center gap-2 text-nude-400 hover:bg-white/40 text-[11px] font-label uppercase tracking-wider transition-all border-r border-border/50";
            
            graphIndicator.classList.remove('hidden');
            codeIndicator.classList.add('hidden');
        }
    }

    let isTerminalExpanded = true;
    function toggleTerminal() {
        const panel = document.getElementById('terminal-panel');
        const icon = document.getElementById('terminal-toggle-icon');
        const content = document.getElementById('terminal-content');
        
        if (isTerminalExpanded) {
            panel.style.height = '32px';
            icon.innerText = 'keyboard_arrow_up';
            content.style.opacity = '0';
        } else {
            panel.style.height = '160px';
            icon.innerText = 'keyboard_arrow_down';
            content.style.opacity = '1';
        }
        isTerminalExpanded = !isTerminalExpanded;
    }

    // Interactions
    document.querySelectorAll('button, .cursor-pointer').forEach(el => {
        el.addEventListener('mousedown', () => {
            el.style.transform = 'scale(0.97)';
        });
        el.addEventListener('mouseup', () => {
            el.style.transform = '';
        });
        el.addEventListener('mouseleave', () => {
            el.style.transform = '';
        });
    });

    const tx = document.querySelector('textarea');
    if (tx) {
        tx.addEventListener('input', function() {
            this.style.height = 'auto';
            this.style.height = Math.min(this.scrollHeight, 120) + 'px';
        });
    }

    // Handle initial prompt enter key
    document.getElementById('initial-prompt').addEventListener('keypress', (e) => {
        if (e.key === 'Enter') enterWorkspace();
    });
</script>
